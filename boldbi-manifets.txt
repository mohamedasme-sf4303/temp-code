# ====================================================
# Optional PostgreSQL Installation
# ====================================================

RUN if [ "$INSTALL_POSTGRES" = "true" ]; then \
      echo "Installing PostgreSQL 15..." && \
      curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc | gpg --dearmor -o /usr/share/keyrings/postgresql.gpg && \
      echo "deb [signed-by=/usr/share/keyrings/postgresql.gpg] http://apt.postgresql.org/pub/repos/apt bookworm-pgdg main" > /etc/apt/sources.list.d/pgdg.list && \
      apt-get update && \
      apt-get install -y postgresql-15 postgresql-contrib-15 && \
      service postgresql start && \
      su - postgres -c "psql -c \"ALTER USER postgres WITH PASSWORD 'Admin@123';\"" && \
      service postgresql stop && \
      apt-get clean; \
    else \
      echo "Skipping PostgreSQL installation"; \
    fi

RUN apt-get purge -y --auto-remove linux-libc-dev || true
RUN rm -rf /var/lib/apt/lists/*

# ====================================================
# Environment Variables
# ====================================================

ENV OPENSSL_CONF=/etc/ssl/
ENV BOLD_SERVICES_HOSTING_ENVIRONMENT=docker
ENV OPTIONAL_LIBS=mongodb,mysql,influxdb,snowflake,oracle,clickhouse,google
ENV INSTALL_POSTGRES=${INSTALL_POSTGRES}

WORKDIR /application

# ====================================================
# Install .NET Runtime 10
# ====================================================

RUN case "$TARGETARCH" in \
      amd64) RUNTIME_ARCH=x64 ;; \
      arm64) RUNTIME_ARCH=arm64 ;; \
      *) echo "Unsupported architecture: $TARGETARCH"; exit 1 ;; \
    esac && \
    mkdir -p /application/dotnet && \
    curl -L -o /tmp/aspnetcore-runtime.tar.gz \
    "https://builds.dotnet.microsoft.com/dotnet/aspnetcore/Runtime/10.0.10/aspnetcore-runtime-10.0.10-linux-${RUNTIME_ARCH}.tar.gz" && \
    tar -xzf /tmp/aspnetcore-runtime.tar.gz -C /application/dotnet && \
    chmod +x /application/dotnet/dotnet && \
    rm -f /tmp/aspnetcore-runtime.tar.gz

RUN ln -sf /application/dotnet/dotnet /usr/bin/dotnet

EXPOSE 80
EXPOSE 443

COPY /application .

# ====================================================
# AMD64 Chrome Setup
# ====================================================

RUN if [ "$TARGETARCH" = "amd64" ]; then \
      cd /application/bi/dataservice/puppeteer/Chrome && \
      unzip chrome-linux64.zip && \
      cp -r chrome-linux64/. . && \
      rm -rf chrome-linux64 chrome-linux64.zip && \
      chmod -R 777 /application/bi/dataservice/puppeteer/Chrome; \
    else \
      echo "ARM64 build - using system Chromium"; \
    fi

RUN chmod +x ./entrypoint.sh

CMD ["./entrypoint.sh"]
